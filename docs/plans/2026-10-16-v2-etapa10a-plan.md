## Global Constraints

- Menú lateral azul oscuro sin cambios.
- Encabezado: título + frase corta + acciones a la derecha (primaria al final).
- Pestañas: "Tablero · Lista · Tabla" con ícono y línea debajo de la activa; accesibles como grupo de radios con nombre "Vista".
- "Filtros" (ícono embudo) → panel con: Mostrar, Prioridad, Proyecto (solo en Mis tareas), "Solo vencidas", Ordenar por, "Limpiar filtros", "Listo". Con filtros activos: "Filtros · N" (N = prioridad ≠ todas, proyecto ≠ todos, solo vencidas).
- Vista por defecto: **Tablero** (preferencia nueva `matecode.view.v2`, así todos arrancan en el tablero).
- Sin tarjetas de números. Conteo en el título de cada columna.
- Tarjeta: número + prioridad; título; descripción 2 líneas; subtareas "N/M" + barra (si tiene); "Asignada a" + cara + selector de estado; pie: fecha (roja si vencida) y proyecto. Hechas: tachadas y en gris.
- "+ Agregar tarea" punteado al pie de Pendiente y En curso (crea con ese estado), solo si puede crear.
- Textos de Mis tareas: título "Mis tareas", frase "Lo que tienes asignado y lo que mueve el equipo, en un solo tablero."
- Sin hex en componentes; contraste ≥ 4.5:1.
- `npm test`, `npm run build`, `npm run lint` antes de cada commit; un commit por tarea con `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```tsx
// src/components/ui/ViewTabs.tsx
import type { LucideIcon } from 'lucide-react';
import styles from './ViewTabs.module.css';

type Option<T extends string> = { value: T; label: string; icon?: LucideIcon; count?: number };
type Props<T extends string> = { label: string; name: string; options: Option<T>[]; value: T; onChange: (value: T) => void };

/** Pestañas para cambiar de vista: radios con aspecto de pestaña (línea debajo de la activa). */
export function ViewTabs<T extends string>({ label, name, options, value, onChange }: Props<T>) {
  return (
    <fieldset className={styles.tabs}>
      <legend className="visually-hidden">{label}</legend>
      {options.map(({ value: v, label: text, icon: Icon, count }) => (
        <label key={v} className={`${styles.tab} ${v === value ? styles.active : ''}`}>
          <input type="radio" className="visually-hidden" name={name} value={v} checked={v === value} onChange={() => onChange(v)} />
          {Icon && <Icon size={16} aria-hidden="true" />}
          {text}
          {count !== undefined && <span className={styles.count}>{count}</span>}
        </label>
      ))}
    </fieldset>
  );
}
```

```tsx
// src/components/ui/FiltersPopover.tsx
import { Filter } from 'lucide-react';
import { useEffect, useId, useRef, useState, type ReactNode } from 'react';
import styles from './FiltersPopover.module.css';

type Props = { activeCount: number; onClear: () => void; children: ReactNode };

/** Botón "Filtros" que abre un panel con los filtros; cierra con "Listo", Escape o un clic afuera. */
export function FiltersPopover({ activeCount, onClear, children }: Props) {
  const [open, setOpen] = useState(false);
  const panelId = useId();
  const root = useRef<HTMLDivElement>(null);
  const button = useRef<HTMLButtonElement>(null);

  useEffect(() => {
    if (!open) return;
    const onDown = (e: MouseEvent) => {
      if (root.current && !root.current.contains(e.target as Node)) setOpen(false);
    };
    document.addEventListener('mousedown', onDown);
    return () => document.removeEventListener('mousedown', onDown);
  }, [open]);

  function close() {
    setOpen(false);
    button.current?.focus();
  }

  return (
    <div className={styles.root} ref={root}>
      <button
        ref={button} type="button" className={styles.trigger} aria-expanded={open} aria-controls={panelId}
        onClick={() => setOpen((o) => !o)}
      >
        <Filter size={15} aria-hidden="true" />
        {activeCount > 0 ? `Filtros · ${activeCount}` : 'Filtros'}
      </button>
      {open && (
        <div
          id={panelId} role="dialog" aria-label="Filtros" className={styles.panel}
          onKeyDown={(e) => { if (e.key === 'Escape') close(); }}
        >
          <div className={styles.fields}>{children}</div>
          <div className={styles.footer}>
            <button type="button" className={styles.link} onClick={onClear}>Limpiar filtros</button>
            <button type="button" className={styles.done} onClick={close}>Listo</button>
          </div>
        </div>
      )}
    </div>
  );
}
```

// src/components/tasks/TaskCard.tsx
import { CalendarDays } from 'lucide-react';
import { Avatar } from '@/components/layout/Avatar';
import { ProjectTag } from '@/components/projects/ProjectTag';
import type { ProjectOf } from '@/features/projects/projectOf';
import { subtaskProgress } from '@/features/tasks/subtasks';
import type { NameOf, PersonOf } from '@/features/team/members';
import { useTaskNumberFormat } from '@/hooks/useWorkspaceSettings';
import { PRIORITY_LABEL, type Task, type TaskStatus } from '@/types/task';
import { dueStatus, formatDueDate } from '@/utils/formatDate';
import { StatusSelect } from './StatusSelect';
import styles from './TaskCard.module.css';

type Props = {
  task: Task; nameOf: NameOf; personOf: PersonOf; projectOf: ProjectOf; editable: boolean; today: string;
  onStatusChange: (id: string, status: TaskStatus) => void; onOpen: (task: Task) => void;
};

/** Tarjeta del tablero: lo esencial de una tarea de un vistazo. */
export function TaskCard({ task, nameOf, personOf, projectOf, editable, today, onStatusChange, onOpen }: Props) {
  const formatNumber = useTaskNumberFormat();
  const done = task.status === 'done';
  const assignee = nameOf(task.assigneeId);
  const person = personOf(task.assigneeId);
  const project = projectOf(task.projectId);
  const sub = subtaskProgress(task.subtasks);
  const due = task.dueDate && !done ? dueStatus(task.dueDate, new Date(`${today}T12:00:00`)) : null;
  return (
    <li className={`${styles.card} ${done ? styles.done : ''}`}>
      <div className={styles.top}>
        <span className={styles.number}>{formatNumber(task.number)}</span>
        <span className={`${styles.priority} ${styles[task.priority]}`}>{PRIORITY_LABEL[task.priority]}</span>
      </div>
      {done && <span className="visually-hidden">Hecha: </span>}
      <button type="button" className={styles.title} onClick={() => onOpen(task)}>{task.title}</button>
      {task.description && <p className={styles.description}>{task.description}</p>}
      {sub.total > 0 && (
        <div className={styles.progress}>
          <div className={styles.progressRow}><span>Subtareas</span><span>{sub.done}/{sub.total}</span></div>
          <div
            className={styles.track} role="progressbar" aria-label={`Subtareas de ${task.title}`}
            aria-valuemin={0} aria-valuemax={100} aria-valuenow={Math.round((sub.done / sub.total) * 100)}
          >
            <span style={{ width: `${(sub.done / sub.total) * 100}%` }} />
          </div>
        </div>
      )}
      <div className={styles.who}>
        <span className={styles.whoLabel}>
          Asignada a
          <span title={assignee ?? 'Sin asignar'}>
            <Avatar name={assignee ?? ''} photo={person?.photo} inactive={person ? !person.active : false} muted={done} />
            <span className="visually-hidden">{assignee ?? 'Sin asignar'}</span>
          </span>
        </span>
        <StatusSelect status={task.status} taskTitle={task.title} canEdit={editable} onChange={(s) => onStatusChange(task.id, s)} />
      </div>
      {(task.dueDate || project) && (
        <div className={styles.footer}>
          {task.dueDate ? (
            <span className={`${styles.due} ${due === 'overdue' ? styles.overdue : ''}`}>
              <CalendarDays size={15} aria-hidden="true" />
              {done ? formatDueDate(task.dueDate) : due === 'overdue' ? `Venció ${formatDueDate(task.dueDate)}` : formatDueDate(task.dueDate)}

}
